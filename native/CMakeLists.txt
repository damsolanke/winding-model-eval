# Builds only villa's vc.grid_raycast extension (volume-cartographer/python/vc/grid_raycast.cpp)
# as a standalone nanobind module. The source is used unmodified; it needs no vc_core.
cmake_minimum_required(VERSION 3.20)
project(windeval_grid_raycast LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(DEFINED ENV{VILLA_DIR})
  set(VILLA_DIR_DEFAULT "$ENV{VILLA_DIR}")
else()
  set(VILLA_DIR_DEFAULT "$ENV{HOME}/kaggle-data/vesuvius/villa")
endif()
set(VILLA_DIR "${VILLA_DIR_DEFAULT}" CACHE PATH "villa checkout")
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir
  OUTPUT_VARIABLE NB_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
list(APPEND CMAKE_PREFIX_PATH "${NB_DIR}")
find_package(nanobind CONFIG REQUIRED)
nanobind_add_module(grid_raycast NOMINSIZE "${VILLA_DIR}/volume-cartographer/python/vc/grid_raycast.cpp")
